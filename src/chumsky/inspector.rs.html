10/// A type that receives event hooks when certain parsing actions occur.
11///
12/// If you don't need to receive event hooks, use [`SimpleState`].
13pub trait Inspector<'src, I: Input<'src>> {
14    /// A type the Inspector can use to revert to a previous state.
15    ///
16    /// For implementation reasons, this is required to be `Clone`.
17    type Checkpoint: Clone;
18
19    /// This function is called when a new token is read from the input stream.
20    // impl note: this should be called only when `self.cursor` is updated, not when we only peek at the next token.
21    fn on_token(&mut self, token: &I::Token);
22    /// This function is called when a combinator saves the current state of the parse.
23    fn on_save<'parse>(&self, cursor: &Cursor<'src, 'parse, I>) -> Self::Checkpoint;
24    /// This function is called when a combinator rewinds to an earlier state of the parser.
25    ///
26    /// You can use [`Checkpoint::inspector`] to get back the [`Checkpoint`][Self::Checkpoint]
27    /// you originally created in [`on_save`][Self::on_save].
28    fn on_rewind<'parse>(&mut self, marker: &Checkpoint<'src, 'parse, I, Self::Checkpoint>);
29}
30
31impl<'src, I: Input<'src>> Inspector<'src, I> for () {
32    type Checkpoint = ();
33    #[inline(always)]
34    fn on_token(&mut self, _: &<I as Input<'src>>::Token) {}
35    #[inline(always)]
36    fn on_save<'parse>(&self, _: &Cursor<'src, 'parse, I>) -> Self::Checkpoint {}
37    #[inline(always)]
38    fn on_rewind<'parse>(&mut self, _: &Checkpoint<'src, 'parse, I, Self>) {}
39}

41/// A state type that should be accessible directly from `parser.state()` and has no special behavior.
42///
43/// This wrapper implements the [`Inspector`] trait for you so you don't have to.
44#[derive(Copy, Clone, Default, Debug)]
45pub struct SimpleState<T>(pub T);
46impl<'src, T, I: Input<'src>> Inspector<'src, I> for SimpleState<T> {
47    type Checkpoint = ();
48    #[inline(always)]
49    fn on_token(&mut self, _: &<I as Input<'src>>::Token) {}
50    #[inline(always)]
51    fn on_save<'parse>(&self, _: &Cursor<'src, 'parse, I>) -> Self::Checkpoint {}
52    #[inline(always)]
53    fn on_rewind<'parse>(&mut self, _: &Checkpoint<'src, 'parse, I, Self::Checkpoint>) {}
54}

56impl<T> Deref for SimpleState<T> {
57    type Target = T;

59    fn deref(&self) -> &Self::Target {
60        &self.0
61    }
62}

64impl<T> DerefMut for SimpleState<T> {
65    fn deref_mut(&mut self) -> &mut Self::Target {

76/// A state type that clones and rolls back its contents during a rewind.
77///
78/// This might be useful if you want to use the parser state to, say, count the parsed occurrences of a particular
79/// construct.
80///
81/// Ideally, you should try to have the [`Clone`] implementation be fairly cheap.
82#[derive(Copy, Clone, Default, Debug)]
83pub struct RollbackState<T>(pub T);
84impl<'src, T: Clone, I: Input<'src>> Inspector<'src, I> for RollbackState<T> {
85    type Checkpoint = T;
86    #[inline(always)]
87    fn on_token(&mut self, _: &<I as Input<'src>>::Token) {}
88    #[inline(always)]
89    fn on_save<'parse>(&self, _: &Cursor<'src, 'parse, I>) -> Self::Checkpoint {
90        self.0.clone()
91    }
92    #[inline(always)]
93    fn on_rewind<'parse>(&mut self, cp: &Checkpoint<'src, 'parse, I, Self::Checkpoint>) {
94        self.0 = cp.inspector.clone();
95    }
96}

98impl<T> Deref for RollbackState<T> {
99    type Target = T;
100
101    fn deref(&self) -> &Self::Target {
102        &self.0
103    }
104}

107    fn deref_mut(&mut self) -> &mut Self::Target {
108        &mut self.0
109    }
110}

118/// A state type that encapsulates a vector, truncating the vector to its original size during a rewind.
119///
120/// This might be useful for representing, say, an arena-style allocator.
121#[derive(Clone, Default, Debug)]
122pub struct TruncateState<T>(pub Vec<T>);
123impl<'src, T, I: Input<'src>> Inspector<'src, I> for TruncateState<T> {
124    type Checkpoint = usize;
125    #[inline(always)]
126    fn on_token(&mut self, _: &<I as Input<'src>>::Token) {}
127    #[inline(always)]
128    fn on_save<'parse>(&self, _: &Cursor<'src, 'parse, I>) -> Self::Checkpoint {
129        self.0.len()
130    }
131    #[inline(always)]
132    fn on_rewind<'parse>(&mut self, cp: &Checkpoint<'src, 'parse, I, Self::Checkpoint>) {
133        self.0.truncate(cp.inspector);
134    }
135}
