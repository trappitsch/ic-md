5/// # Safety
6/// Implers of this trait will get their memory changed through a byte slice.
7/// So make sure the type has a stable ABI and has no padding bits.
8pub unsafe trait Fieldset: Sized {
9    /// Metadata describing some properties of the fieldset
10    const METADATA: FieldsetMetadata;
11    /// Get a zero-initialized instance of the fieldset
12    const ZERO: Self;
13
14    /// Get the fieldset as a mutable slice
15    fn as_slice_mut(&mut self) -> &mut [u8] {
16        unsafe {
17            core::slice::from_raw_parts_mut(
18                (self as *mut Self).cast::<u8>(),
19                core::mem::size_of::<Self>(),
20            )
21        }
22    }
23}

25unsafe impl<T: Fieldset, const N: usize> Fieldset for [T; N] {
26    const METADATA: FieldsetMetadata = T::METADATA;
27    const ZERO: Self = [T::ZERO; N];
28}

34/// Metadata about fieldsets
35#[derive(Debug, Clone, PartialEq, Eq, Hash)]
36#[cfg_attr(feature = "defmt", derive(defmt::Format))]
37#[non_exhaustive]
38pub struct FieldsetMetadata {
39    /// The byte order of the fieldset
40    pub byte_order: ByteOrder,
41}
42
43impl FieldsetMetadata {
44    /// A default that allows you to construct the metadata in const contexts.
45    /// Yields the same value as [`Self::default`].
46    pub const DEFAULT: Self = Self {
47        byte_order: ByteOrder::LE,
48    };
49
50    /// Create a new instance with the default value
51    pub const fn new() -> Self {
52        Self::DEFAULT
53    }
54
55    /// Set the byte order
56    pub const fn with_byte_order(self, byte_order: ByteOrder) -> Self {
57        Self { byte_order, ..self }
58    }
59}
60
61impl Default for FieldsetMetadata {
62    fn default() -> Self {
63        Self::DEFAULT
64    }
65}

87macro_rules! impl_append {
88    ($name:ident -> None, $(($tname:ident: $tnum:tt)),+) => {
89        // No append impl
90    };
91    ($name:ident -> $next:ident, $(($tname:ident: $tnum:tt)),+) => {
92        impl<$($tname),*, NEW> Append<NEW> for $name<$($tname),*> {
93            type Appended = $next<$($tname),* , NEW>;
94            fn append(self, val: NEW) -> Self::Appended {
95                $next($(self.$tnum),*, val)
96            }
97        }
98    };
99}

101macro_rules! create_fsn {
102    ($name:ident -> $next:ident, $(($tname:ident: $tnum:tt)),+) => {
103        /// Combined fieldsets
104        #[derive(Debug)]
105        #[repr(C)]
106        #[doc(hidden)]
107        pub struct $name<$($tname),*>($($tname),*);
108
109        unsafe impl<$($tname: Fieldset),*> Fieldset for $name<$($tname),*> {
110            const METADATA: FieldsetMetadata = A::METADATA;
111            const ZERO: Self = Self($($tname::ZERO),*);
112        }
113
114        impl<$($tname),*> ToTuple for $name<$($tname),*> {
115            type Tuple = ($($tname),*,);
116            fn to_tuple(self) -> Self::Tuple {
117                (
118                    $(self.$tnum),*,
119                )
120            }
121        }
122        impl<'a, $($tname),*> ToTuple for &'a mut $name<$($tname),*> {
123            type Tuple = ($(&'a mut $tname),*,);
124            fn to_tuple(self) -> Self::Tuple {
125                (
126                    $(&mut self.$tnum),*,
127                )
128            }
129        }
130
131        impl_append!($name -> $next, $(($tname: $tnum)),+);
132    }
133}
