3use device_driver_common::specifiers::Access;
4use device_driver_diagnostics::{Diagnostics, DynError, errors::UnspecifiedAccess};

11use super::Assumption;

13pub struct AccessSet;
14
15impl Pass for AccessSet {
16    const ASSUMPTIONS_MADE: &[Assumption] = &[];
17    const ASSUMPTIONS_RELEASED: &[Assumption] = &[Assumption::AccessSet];
18
19    fn run_pass(
20        manifest: &mut Manifest,
21        diagnostics: &mut Diagnostics,
22    ) -> Result<HashSet<UniqueId>, DynError> {
23        set_access(manifest.default_access, &mut manifest.objects, diagnostics);
24        Ok(Default::default())
25    }
26}
27
28fn set_access(
29    default_access: Option<Access>,
30    objects: &mut [Object],
31    diagnostics: &mut Diagnostics,
32) {
33    for object in objects {
34        match object {
35            Object::Device(device) => {
36                let default_access = device.default_access.or(default_access);
37                set_access(default_access, &mut device.objects, diagnostics);
38            }
39            Object::Block(block) => {
40                let default_access = block.default_access.or(default_access);
41                set_access(default_access, &mut block.objects, diagnostics);
42            }
43            Object::FieldSet(field_set) => {
44                let default_access = field_set.default_access.or(default_access);
45                for field in field_set.fields.iter_mut() {
46                    field.access = field.access.or(default_access);
47
48                    if field.access.is_none() {
49                        field.access = Some(Access::RW);
50                        diagnostics.add(UnspecifiedAccess {
51                            object_name: field.name.span,
52                            short_property: true,
53                            properties_span: Some(field.short_properties_span),
54                        });
55                    }
56                }
57            }
58            Object::Register(register) => {
59                register.access = register.access.or(default_access);
60
61                if register.access.is_none() {
62                    register.access = Some(Access::RW);
63                    diagnostics.add(UnspecifiedAccess {
64                        object_name: register.name.span,
65                        short_property: false,
66                        properties_span: register.properties_span,
67                    });
68                }
69            }
70            Object::Buffer(buffer) => {
71                buffer.access = buffer.access.or(default_access);
72
73                if buffer.access.is_none() {
74                    buffer.access = Some(Access::RW);
75                    diagnostics.add(UnspecifiedAccess {
76                        object_name: buffer.name.span,
77                        short_property: false,
78                        properties_span: buffer.properties_span,
79                    });
80                }
81            }
82
83            Object::Field(_) => {
84                // Intentionally left empty as fields are done inline in the fieldset case
85            }
86            Object::Command(_) | Object::Enum(_) | Object::Extern(_) => {
87                // Intentionally left empty as they don't have children we care about and they don't carry an access specifier themselves
88            }
89        }
90    }
91}
