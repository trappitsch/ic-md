22/// Checks if enums are fully specified and determines the generation style
23pub struct EnumValuesChecked;
24
25impl Pass for EnumValuesChecked {
26    const ASSUMPTIONS_MADE: &[Assumption] = &[];
27    const ASSUMPTIONS_RELEASED: &[Assumption] = &[
28        Assumption::EnumBaseTypesSpecified,
29        Assumption::EnumsNotEmpty,
30    ];
31
32    fn run_pass(
33        manifest: &mut Manifest,
34        diagnostics: &mut Diagnostics,
35    ) -> Result<HashSet<UniqueId>, DynError> {
36        let mut removals = HashSet::new();
37
38        let mut iter = manifest.iter_objects_with_config_mut();
39        while let Some((object, _)) = iter.next() {
40            let Object::Enum(enum_value) = object else {
41                continue;
42            };
43
44            if enum_value.variants.is_empty() {
45                diagnostics.add(EmptyEnum {
46                    enum_node: enum_value.span,
47                });
48                removals.insert(enum_value.id());
49                continue;
50            }
51
52            // Record all variant values
53            let e_id = enum_value.id();
54            let seen_values = enum_value
55                .iter_variants_with_discriminant_mut()
56                .map(|(discriminant, variant)| {
57                    if variant.value.specified_discriminant().is_none() {
58                        variant.value.specify(discriminant);
59                    }
60                    (discriminant, (variant.id_with(e_id.clone()), variant.span))
61                })
62                .collect_vec();
63
64            let mut seen_values_map = HashMap::<i128, Vec<(&UniqueId, Span)>>::new();
65            for (variant_value, (variant_id, span)) in &seen_values {
66                seen_values_map
67                    .entry(*variant_value)
68                    .or_default()
69                    .push((variant_id, *span));
70            }
71            for (value, variants) in seen_values_map
72                .into_iter()
73                .sorted_unstable_by_key(|(value, _)| *value)
74            {
75                if variants.len() > 1 {
76                    diagnostics.add(DuplicateVariantValue {
77                        duplicates: variants.iter().map(|(_, span)| *span).collect(),
78                        value,
79                    });
80                    removals.insert(enum_value.id());
81                }
82            }
83
84            let (seen_min, (_, seen_min_span)) = seen_values
85                .iter()
86                .min_by_key(|(val, _)| val)
87                .expect("Enums must not be empty");
88            let (seen_max, _) = seen_values
89                .iter()
90                .max_by_key(|(val, _)| val)
91                .expect("Enums must not be empty");
92
93            let base_type_integer = match enum_value.base_type.value {
94                BaseType::Unspecified => Integer::find_smallest(
95                    *seen_min,
96                    *seen_max,
97                    enum_value.size_bits.unwrap_or_default().into(),
98                ),
99                BaseType::Bool => {
100                    diagnostics.add(EnumBadBasetype {
101                        enum_name: enum_value.name.span,
102                        base_type: enum_value.base_type.span,
103                        info: "all enums must have an integer as base type",
104                        context: vec![],
105                    });
106                    removals.insert(enum_value.id());
107                    continue;
108                }
109                BaseType::Uint => {
110                    let integer = Integer::find_smallest(
111                        *seen_min,
112                        *seen_max,
113                        enum_value.size_bits.unwrap_or_default().into(),
114                    );
115
116                    if integer.is_some_and(|i| i.is_signed()) {
117                        // TODO: Make separate error type (same as below)
118                        diagnostics.add(EnumBadBasetype {
119                        enum_name: enum_value.name.span,
120                        base_type: enum_value.base_type.span,
121                        info: "enums must use a signed integer when any variant has a negative value",
122                        context: vec![format!("variant with negative value: {seen_min}").with_span(*seen_min_span)],
123                    });
124                        removals.insert(enum_value.id());
125                        continue;
126                    }
127
128                    integer
129                }
130                BaseType::Int => Integer::find_smallest(
131                    (*seen_min).min(-1),
132                    *seen_max,
133                    enum_value.size_bits.unwrap_or_default().into(),
134                ),
135                BaseType::FixedSize(integer) => {
136                    if enum_value.size_bits.unwrap_or_default() > integer.size_bits() {
137                        diagnostics.add(EnumSizeBitsBiggerThanBaseType {
138                            enum_name: enum_value.name.span,
139                            base_type: enum_value.base_type.span,
140                            enum_size_bits: enum_value.size_bits.unwrap_or_default(),
141                            base_type_size_bits: integer.size_bits(),
142                        });
143                        removals.insert(enum_value.id());
144                        continue;
145                    }
146
147                    if !integer.is_signed() && seen_min.is_negative() {
148                        // TODO: Make separate error type (same as above)
149                        diagnostics.add(EnumBadBasetype {
150                        enum_name: enum_value.name.span,
151                        base_type: enum_value.base_type.span,
152                        info: "enums must use a signed integer when any variant has a negative value",
153                        context: vec![format!("variant with negative value: {seen_min}").with_span(*seen_min_span)],
154                    });
155                        removals.insert(enum_value.id());
156                        continue;
157                    }
158
159                    Some(integer)
160                }
161            };
162
163            let Some(base_type_integer) = base_type_integer else {
164                diagnostics.add(EnumNoAutoBaseTypeSelected {
165                    enum_name: enum_value.name.span,
166                });
167                removals.insert(enum_value.id());
168                continue;
169            };
170
171            let size_bits = match enum_value.size_bits {
172                None => base_type_integer
173                    .bits_required(*seen_min, *seen_max)
174                    .min(base_type_integer.size_bits()),
175                Some(size_bits) => size_bits,
176            };
177
178            enum_value.base_type.value = BaseType::FixedSize(base_type_integer);
179            enum_value.size_bits = Some(size_bits);
180
181            let all_values = if base_type_integer.is_signed() {
182                let max = (1 << size_bits) / 2;
183                -max..=max - 1
184            } else {
185                0..=(1 << size_bits) - 1
186            };
187
188            // Check if all bits are covered or if there's a fallback variant
189            let has_fallback = enum_value
190                .variants
191                .iter()
192                .any(|v| matches!(v.value, EnumValue::Default(_) | EnumValue::CatchAll(_)));
193            let has_bits_covered = all_values
194                .clone()
195                .all(|val| seen_values.iter().any(|(seen_val, _)| val == *seen_val));
196
197            enum_value.generation_style = Some(match (has_fallback, has_bits_covered) {
198                (true, _) => EnumGenerationStyle::Fallback,
199                (false, true) => EnumGenerationStyle::InfallibleWithinRange,
200                (false, false) => EnumGenerationStyle::Fallible,
201            });
202
203            // Check if the enum has variants that fall outside of the available bits
204            let too_high_values = seen_values
205                .iter()
206                .filter(|(val, _)| val > all_values.end())
207                .map(|(_, (_, span))| *span)
208                .collect::<Vec<_>>();
209            let too_low_values = seen_values
210                .iter()
211                .filter(|(val, _)| val < all_values.start())
212                .map(|(_, (_, span))| *span)
213                .collect::<Vec<_>>();
214
215            if !too_high_values.is_empty() {
216                diagnostics.add(VariantValuesTooHigh {
217                    variant_names: too_high_values,
218                    enum_name: enum_value.name.span,
219                    max_value: *all_values.end(),
220                    size_bits,
221                });
222                removals.insert(enum_value.id());
223                continue;
224            }
225            if !too_low_values.is_empty() {
226                diagnostics.add(VariantValuesTooLow {
227                    variant_names: too_low_values,
228                    enum_name: enum_value.name.span,
229                    min_value: *all_values.start(),
230                    size_bits,
231                });
232                removals.insert(enum_value.id());
233                continue;
234            }
235
236            // Check whether the enum has more than one default
237            let default_variants = enum_value
238                .variants
239                .iter()
240                .filter(|v| v.value.is_default())
241                .map(|v| v.span)
242                .collect::<Vec<_>>();
243
244            if default_variants.len() > 1 {
245                diagnostics.add(EnumMultipleDefaults {
246                    variant_names: default_variants,
247                    enum_name: enum_value.name.span,
248                });
249
250                // Set all but the first default to unspecified. This aids keeping the generated code available
251                let mut defaults_seen = 0;
252                for variant in &mut enum_value.variants {
253                    if variant.value.is_default() {
254                        if defaults_seen != 0 {
255                            variant.value = EnumValue::Unspecified;
256                        }
257                        defaults_seen += 1;
258                    }
259                }
260            }
261
262            // Check whether the enum has more than one catch all
263            let catch_all_variants = enum_value
264                .variants
265                .iter()
266                .filter(|v| v.value.is_catch_all())
267                .map(|v| v.span)
268                .collect::<Vec<_>>();
269
270            if catch_all_variants.len() > 1 {
271                diagnostics.add(EnumMultipleCatchalls {
272                    variant_names: catch_all_variants,
273                    enum_name: enum_value.name.span,
274                });
275
276                // Set all but the first catch-all to unspecified. This aids keeping the generated code available
277                let mut catch_alls_seen = 0;
278                for variant in &mut enum_value.variants {
279                    if variant.value.is_catch_all() {
280                        if catch_alls_seen != 0 {
281                            variant.value = EnumValue::Unspecified;
282                        }
283                        catch_alls_seen += 1;
284                    }
285                }
286            }
287        }
288
289        Ok(removals)
290    }
291}
