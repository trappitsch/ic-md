13impl<'a> Generator<'a> {
14    pub fn impl_fork(
15        &mut self,
16        state: State,
17        state_data: &StateData,
18        ignore_self: bool,
19    ) -> TokenStream {
20        if state_data.normal.len() > 2 {
21            self.impl_fork_table(state, state_data, ignore_self)
22        } else {
23            self.impl_fork_match(state, state_data, ignore_self)
24        }
25    }
26
27    // Generate code for encountering the end of input.
28    // If we are not in the middle of a token, return None (the iterator is ended)
29    // If the state has an EOI node, transition to it.
30    // Otherwise, fall through to later code.
31    fn fork_eoi(&self, state: State, state_data: &StateData) -> TokenStream {
32        let mut eoi = TokenStream::new();
33        if state == self.graph.root() {
34            // If we just started lexing and are at the end of input, return None
35            eoi.append_all(quote! { if lex.offset() == offset { return None } });
36        }
37        if let &Some(eoi_state) = &state_data.eoi {
38            let transition = self.state_transition(eoi_state);
39            eoi.append_all(quote! {
40                offset += 1;
41                #transition
42            });
43        }
44
45        eoi
46    }

48    fn impl_fork_match(
49        &mut self,
50        state: State,
51        state_data: &StateData,
52        ignore_self: bool,
53    ) -> TokenStream {
54        // Generate a match arm for each byte class, with each body being a state transition
55        let mut inner_cases = TokenStream::new();
56        for (byte_class, next_state) in &state_data.normal {
57            if ignore_self && next_state == &state {
58                continue;
59            }
60
61            let comparisons = byte_class.impl_with_cmp();
62            let cmp_count: usize = comparisons.iter().map(|cmp| cmp.count_ops()).sum();
63
64            let condition = if cmp_count > 2 {
65                let (test_ident, test_mask) = self.add_test_to_lut(byte_class);
66                quote! { #test_ident[byte as usize] & #test_mask != 0 }
67            } else {
68                let sub_conditions = comparisons
69                    .into_iter()
70                    .map(|cmp| {
71                        let Comparisons { range, except } = cmp;
72                        let start = byte_to_tokens(*range.start());
73                        let end = byte_to_tokens(*range.end());
74                        let exceptions = except
75                            .into_iter()
76                            .map(|ex| {
77                                quote! { && byte != #ex }
78                            })
79                            .collect::<Vec<_>>();
80                        if range.len() == 1 {
81                            quote! { (byte == #start) }
82                        } else {
83                            quote! { (matches!(byte, #start ..= #end) #(#exceptions)*) }
84                        }
85                    })
86                    .collect::<Vec<_>>();
87
88                quote! { #(#sub_conditions) ||* }
89            };
90            let transition = self.state_transition(*next_state);
91            inner_cases.append_all(quote! {
92                if #condition {
93                    offset += 1;
94                    #transition
95                }
96            });
97        }
98
99        let eoi = self.fork_eoi(state, state_data);
100        quote! {
101            let other = lex.read::<u8>(offset);
102            if let Some(byte) = other {
103                #inner_cases
104            } else {
105                #eoi
106            }
107            _take_action!(lex, offset, context, state)
108        }
109    }

111    fn impl_fork_table(
112        &mut self,
113        state: State,
114        state_data: &StateData,
115        ignore_self: bool,
116    ) -> TokenStream {
117        // Generate a match arm for each byte class, with each body being a state transition
118        let mut table = vec![None; 256];
119        for (byte_class, next_state) in &state_data.normal {
120            if ignore_self && next_state == &state {
121                continue;
122            }
123
124            for range in &byte_class.ranges {
125                for byte in range.clone() {
126                    table[byte as usize] = Some(*next_state);
127                }
128            }
129        }
130
131        // We need this distinction because the state machine states can be stored directly in the
132        // table, while the function calls need a table of enums followed by a match to satisfy the
133        // borrow checker (the state function signatures contain lifetimes).
134        let body = if self.config.use_state_machine_codegen {
135            let table_elements = table
136                .into_iter()
137                .map(|state_op| match state_op {
138                    Some(state) => {
139                        let val = self.state_value(state);
140                        quote!(Some(#val))
141                    }
142                    None => quote!(None),
143                })
144                .collect::<Vec<_>>();
145
146            let action = self.state_action(quote!(next_state));
147            quote! {
148                const TABLE: [_Option<LogosState>; 256] = [#(#table_elements),*];
149                let next_state = TABLE[byte as usize];
150                if let Some(next_state) = next_state {
151                    offset += 1;
152                    #action
153                }
154            }
155        } else {
156            let states_set = table.iter().filter_map(|&op| op).collect::<HashSet<_>>();
157            let mut states = states_set.into_iter().collect::<Vec<_>>();
158            // Sort for generated source stability
159            states.sort_unstable();
160
161            let idents = states
162                .iter()
163                .map(|state| &self.state_idents[state][1])
164                .collect::<Vec<_>>();
165
166            let actions = states
167                .iter()
168                .cloned()
169                .map(|state| self.state_transition(state))
170                .collect::<Vec<_>>();
171
172            // Explicit fallthrough to otherwise case later in the function
173            let match_body = quote! {
174                #(LogosNextState::#idents => { #actions }),*
175                LogosNextState::___ => {},
176            };
177
178            let table_elements = table
179                .into_iter()
180                .map(|state_op| match state_op {
181                    Some(state) => self.state_idents[&state][1].to_token_stream(),
182                    None => quote!(___),
183                })
184                .collect::<Vec<_>>();
185
186            // Undo the unconditional increment if we don't have a next state
187            quote! {
188                #[derive(Copy, Clone)]
189                enum LogosNextState {
190                    ___,
191                    #(#idents),*
192                }
193                const TABLE: [LogosNextState; 256] = {
194                    use LogosNextState::*;
195                    [ #(#table_elements),* ]
196                };
197                offset += 1;
198                match TABLE[byte as usize] {
199                    #match_body
200                }
201                offset -= 1;
202            }
203        };
204
205        let eoi = self.fork_eoi(state, state_data);
206
207        quote! {
208            let other = lex.read::<u8>(offset);
209            if let Some(byte) = other {
210                #body
211            } else {
212                #eoi
213            }
214            _take_action!(lex, offset, context, state)
215        }
216    }
