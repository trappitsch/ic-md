10pub type OwnedDFA = DFA<Vec<u32>>;

12/// Returns an iterator over the matching patterns of a given dfa state. Returns leaf ids in
13/// ascending order.
14pub fn iter_matches<'a>(state_id: StateID, dfa: &'a OwnedDFA) -> impl Iterator<Item = LeafId> + 'a {
15    let num_matches = if dfa.is_match_state(state_id) {
16        dfa.match_len(state_id)
17    } else {
18        0
19    };
20
21    (0..num_matches).map(move |match_idx| {
22        let pattern_id = dfa.match_pattern(state_id, match_idx);
23        LeafId::from(pattern_id)
24    })
25}
26
27/// Returns an iterator over the child states of a given dfa state. Returns children in order of
28/// input byte `(0..=255)`, then eoi. No deduplication of child states is performed.
29pub fn iter_children<'a>(dfa: &'a OwnedDFA, state: StateID) -> impl Iterator<Item = StateID> + 'a {
30    (0..=u8::MAX)
31        .map(move |byte| dfa.next_state(state, byte))
32        .chain(iter::once(dfa.next_eoi_state(state)))
33}
34
35/// This utility function returns every state accessible by the dfa
36/// from a root state. Returns the states in ascending order.
37pub fn get_states(dfa: &OwnedDFA, root: StateID) -> impl Iterator<Item = StateID> {
38    let mut states = HashSet::new();
39    states.insert(root);
40    let mut explore_stack = vec![root];
41    while let Some(state) = explore_stack.pop() {
42        for child in iter_children(dfa, state) {
43            if states.insert(child) {
44                explore_stack.push(child);
45            }
46        }
47    }
48
49    let mut sorted = states.into_iter().collect::<Vec<_>>();
50    sorted.sort_unstable();
51    sorted.into_iter()
52}
