6    hir::{Dot, Hir, HirKind},

10static DOT_HIRS: LazyLock<[Hir; 6]> = LazyLock::new(|| {
11    [
12        Hir::dot(Dot::AnyChar),
13        Hir::dot(Dot::AnyByte),
14        Hir::dot(Dot::AnyByteExceptLF),
15        Hir::dot(Dot::AnyCharExceptLF),
16        Hir::dot(Dot::AnyByteExceptCRLF),
17        Hir::dot(Dot::AnyCharExceptCRLF),
18    ]
19});

21#[derive(Clone, Debug)]
22pub struct Pattern {
23    /// This field is only used to display #[regex] or #[token] in the display impl
24    is_literal: bool,
25    /// The original source literal for this pattern
26    source: String,
27    /// The parsed regex HIR for this pattern
28    hir: Hir,
29}
30
31impl Pattern {
32    /// Create a new pattern from a regex string source.
33    ///
34    /// # Arguments
35    ///
36    /// - `unicode`: whether the regex pattern should match bytes (false), or utf8 codepoints (true)
37    /// - `ignore_case`: whether to set the `(?i)` flag for the entire pattern.
38    pub fn compile(
39        is_literal: bool,
40        regex: &str,
41        source: String,
42        unicode: bool,
43        ignore_case: bool,
44    ) -> Result<Pattern, String> {
45        // UTF-8 mode is disabled here so we can give prettier error messages
46        // later in the compilation process. See logos_codegen/src/lib.rs for
47        // the utf8 checking.
48        let hir = ParserBuilder::new()
49            .utf8(false)
50            .unicode(unicode)
51            .case_insensitive(ignore_case)
52            .build()
53            .parse(regex)
54            .map_err(|err| format!("{err}"))?;
55
56        Ok(Pattern {
57            is_literal,
58            source,
59            hir,
60        })
61    }
62
63    /// Create a pattern that matches a literal.
64    ///
65    /// This function avoids escaping by constructing an `Hir::literal` directly.
66    pub fn compile_lit(source: &Literal) -> Result<Pattern, String> {
67        let hir = match source {
68            Literal::Utf8(lit_str) => Hir::literal(lit_str.value().as_bytes()),
69            Literal::Bytes(lit_byte_str) => Hir::literal(lit_byte_str.value()),
70        };
71
72        Ok(Pattern {
73            is_literal: true,
74            source: source.token().to_string(),
75            hir,
76        })
77    }
78
79    /// Get the default priority for a pattern
80    pub fn priority(&self) -> usize {
81        Self::complexity(&self.hir)
82    }
83
84    fn complexity(hir: &Hir) -> usize {
85        match hir.kind() {
86            HirKind::Empty => 0,
87            HirKind::Literal(lit) => match std::str::from_utf8(&lit.0) {
88                Ok(s) => 2 * s.chars().count(),
89                Err(_) => 2 * lit.0.len(),
90            },
91            HirKind::Class(_) => 2,
92            HirKind::Look(_) => 0,
93            HirKind::Repetition(repetition) => {
94                repetition.min as usize * Self::complexity(&repetition.sub)
95            }
96            HirKind::Capture(capture) => Self::complexity(&capture.sub),
97            HirKind::Concat(hirs) => hirs.iter().map(Self::complexity).sum(),
98            HirKind::Alternation(hirs) => hirs.iter().map(Self::complexity).min().unwrap_or(0),
99        }
100    }
101
102    /// Return true if this pattern contains a non-greedy `.+` or `.*`
103    pub fn check_for_greedy_all(&self) -> bool {
104        Self::has_greedy_all(&self.hir)
105    }
106
107    fn has_greedy_all(hir: &Hir) -> bool {
108        match hir.kind() {
109            HirKind::Repetition(repetition) => {
110                let is_dot = DOT_HIRS.contains(&repetition.sub);
111                let is_unbounded = repetition.max.is_none();
112                let is_greedy = repetition.greedy;
113
114                is_dot && is_unbounded && is_greedy
115            }
116            HirKind::Empty => false,
117            HirKind::Literal(_literal) => false,
118            HirKind::Class(_class) => false,
119            HirKind::Look(_look) => false,
120            HirKind::Capture(capture) => Self::has_greedy_all(&capture.sub),
121            HirKind::Concat(hirs) => hirs.iter().any(Self::has_greedy_all),
122            HirKind::Alternation(hirs) => hirs.iter().any(Self::has_greedy_all),
123        }
124    }
125
126    /// Get a reference to a pattern's Hir
127    pub fn hir(&self) -> &Hir {
128        &self.hir
129    }
130
131    /// Get a reference to the original source string of the pattern
132    pub fn source(&self) -> &str {
133        &self.source
134    }
135}
136
137impl fmt::Display for Pattern {
138    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
139        if self.is_literal {
140            write!(f, "#[token({})]", self.source)
141        } else {
142            write!(f, "#[regex({})]", self.source)
143        }
144    }
